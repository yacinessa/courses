Exercise 2: Traffic Light Simulation

Objective:

Create a traffic light simulation with 3 LEDs (Red, Yellow, Green) that light up according to a realistic cycle.

Sequence: Red (5s) → Red + Yellow (2s) → Green (5s) → Yellow (2s) → repeat

STM32 C Code:

// GPIO Configuration
// LED Red:   GPIOA, Pin 0
// LED Yellow: GPIOA, Pin 1  
// LED Green:  GPIOA, Pin 2

#include "stm32f1xx.h"

void setRed() {
    GPIOA->ODR |= (1 << 0);    // Red ON
    GPIOA->ODR &= ~((1 << 1) | (1 << 2)); // Yellow & Green OFF
}

void setRedYellow() {
    GPIOA->ODR |= ((1 << 0) | (1 << 1));  // Red & Yellow ON
    GPIOA->ODR &= ~(1 << 2);               // Green OFF
}

void setGreen() {
    GPIOA->ODR |= (1 << 2);    // Green ON
    GPIOA->ODR &= ~((1 << 0) | (1 << 1)); // Red & Yellow OFF
}

void setYellow() {
    GPIOA->ODR |= (1 << 1);    // Yellow ON
    GPIOA->ODR &= ~((1 << 0) | (1 << 2)); // Red & Green OFF
}

void allOff() {
    GPIOA->ODR &= ~((1 << 0) | (1 << 1) | (1 << 2)); // All OFF
}

int main(void) {
    // Initialize GPIOA pins 0,1,2 as outputs
    RCC->APB2ENR |= RCC_APB2ENR_IOPAEN;
    GPIOA->CRL &= ~0xFFF;      // Clear pins 0,1,2
    GPIOA->CRL |= 0x222;       // Push-pull output, 50MHz
    
    while(1) {
        // Red (5 seconds)
        setRed();
        HAL_Delay(5000);
        
        // Red + Yellow (2 seconds)
        setRedYellow();
        HAL_Delay(2000);
        
        // Green (5 seconds)
        setGreen();
        HAL_Delay(5000);
        
        // Yellow (2 seconds)
        setYellow();
        HAL_Delay(2000);
    }
}
