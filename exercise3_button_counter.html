Exercise 3: Button Press Counter

Objective:

Create a counter that increments each time a button is pressed and displays the count in binary using 4 LEDs (4 bits: 0-15).

Hardware Setup:

  • Button: GPIOC, Pin 13 (pull-up internal)
  • LED Bit 0 (LSB): GPIOA, Pin 0
  • LED Bit 1: GPIOA, Pin 1
  • LED Bit 2: GPIOA, Pin 2
  • LED Bit 3 (MSB): GPIOA, Pin 3
  • Count Range: 0-15 (returns to 0 after 15)

STM32 C Code:

#include "stm32f1xx.h"

volatile uint8_t button_count = 0;
volatile uint32_t last_button_time = 0;
volatile uint8_t button_pressed = 0;

// Button debounce function
uint8_t debounce_button(void) {
    static uint32_t last_read = 0;
    static uint8_t button_state = 1; // Pull-up, so idle = 1
    uint8_t current_read;
    
    current_read = HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_13);
    
    if(current_read != button_state) {
        last_read = HAL_GetTick();
        button_state = current_read;
    }
    
    // 50ms debounce time
    if((HAL_GetTick() - last_read) > 50) {
        if(current_read == 0) { // Button pressed (active low)
            return 1;
        }
    }
    return 0;
}

// Display count on LEDs
void display_count(uint8_t count) {
    // Clear all bits first
    GPIOA->ODR &= ~0xF; // Clear bits 0-3
    
    // Set bits according to count (binary)
    GPIOA->ODR |= (count & 0xF);
}

// Get button state with debouncing
uint8_t get_button_pressed(void) {
    if(debounce_button() && !button_pressed) {
        button_pressed = 1;
        return 1;
    } else if(!debounce_button()) {
        button_pressed = 0;
    }
    return 0;
}

int main(void) {
    // Initialize GPIOA pins 0-3 as outputs (LEDs)
    RCC->APB2ENR |= RCC_APB2ENR_IOPAEN;
    GPIOA->CRL &= ~0xFFFF;        // Clear pins 0-3
    GPIOA->CRL |= 0x2222;         // Push-pull output, 50MHz
    
    // Initialize GPIOC pin 13 as input (button)
    RCC->APB2ENR |= RCC_APB2ENR_IOPCEN;
    GPIOA->CRL &= ~(0xF << 20);   // Clear PC13 config
    
    while(1) {
        // Check for button press
        if(get_button_pressed()) {
            button_count++;
            if(button_count > 15) {
                button_count = 0; // Reset after 15
            }
            
            // Update display
            display_count(button_count);
            
            // Simple delay to avoid multiple counts
            HAL_Delay(200);
        }
    }
}
